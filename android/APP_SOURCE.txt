package com.levoy.app

import android.app.Application
import android.app.NotificationChannel
import android.app.NotificationManager
import android.os.Build

class LevoyApp : Application() {

    companion object {
        const val CHANNEL_SYNC = "call_sync_channel"
        const val CHANNEL_ALERT = "call_alert_channel"
    }

    override fun onCreate() {
        super.onCreate()
        createNotificationChannels()
    }

    private fun createNotificationChannels() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val syncChannel = NotificationChannel(
                CHANNEL_SYNC,
                "Call Sync",
                NotificationManager.IMPORTANCE_LOW
            ).apply {
                description = "Background call log synchronization"
                setShowBadge(false)
            }
            val alertChannel = NotificationChannel(
                CHANNEL_ALERT,
                "Call Alerts",
                NotificationManager.IMPORTANCE_HIGH
            ).apply {
                description = "Missed call and important call alerts"
            }
            val manager = getSystemService(NotificationManager::class.java)
            manager.createNotificationChannel(syncChannel)
            manager.createNotificationChannel(alertChannel)
        }
    }
}

package com.levoy.app.network

import android.content.Context
import com.levoy.app.model.*
import com.google.gson.Gson
import com.google.gson.GsonBuilder
import okhttp3.*
import okhttp3.MediaType.Companion.toMediaType
import java.io.File
import java.util.concurrent.TimeUnit

class ApiClient(context: Context) {

    private val gson: Gson = GsonBuilder().setDateFormat("yyyy-MM-dd'T'HH:mm:ss").create()
    private val baseUrl: String = BuildConfig.API_BASE_URL

    private val authInterceptor = Interceptor { chain ->
        val token = TokenStorage.getInstance(context).getToken()
        val request = chain.request().newBuilder()
        if (token != null) {
            request.addHeader("Authorization", "Bearer $token")
        }
        chain.proceed(request.build())
    }

    private val client: OkHttpClient = OkHttpClient.Builder()
        .addInterceptor(authInterceptor)
        .addInterceptor(LoggingInterceptor())
        .connectTimeout(30, TimeUnit.SECONDS)
        .readTimeout(30, TimeUnit.SECONDS)
        .writeTimeout(30, TimeUnit.SECONDS)
        .build()

    // ─── Auth ───

    fun login(email: String, password: String): Result<AuthResponse> {
        return executeRequest<AuthResponse>(
            "POST",
            "/auth/login",
            mapOf("email" to email, "password" to password)
        )
    }

    fun register(email: String, password: String, name: String, phone: String? = null): Result<AuthResponse> {
        val body = mutableMapOf("email" to email, "password" to password, "name" to name)
        phone?.let { body["phone"] = it }
        return executeRequest<AuthResponse>("POST", "/auth/register", body)
    }

    fun getMe(): Result<User> {
        return executeRequest<User>("GET", "/auth/me")
    }

    // ─── Calls ───

    fun uploadCallBatch(events: List<CallEventUpload>): Result<Map<String, Any>> {
        val body = mapOf("events" to events)
        return executeRequest<Map<String, Any>>("POST", "/calls/batch", body)
    }

    fun getCalls(limit: Int = 100, offset: Int = 0, direction: String? = null,
                 status: String? = null, phone: String? = null,
                 dateFrom: String? = null, dateTo: String? = null,
                 pinned: Boolean? = null): Result<List<CallEvent>> {
        val params = mutableMapOf<String, String>(
            "limit" to limit.toString(),
            "offset" to offset.toString()
        )
        direction?.let { params["direction"] = it }
        status?.let { params["status"] = it }
        phone?.let { params["phone"] = it }
        dateFrom?.let { params["date_from"] = it }
        dateTo?.let { params["date_to"] = it }
        if (pinned != null) params["pinned"] = pinned.toString()
        return executeRequest<List<CallEvent>>("GET", "/calls?${params.entries.joinToString("&") { "${it.key}=${it.value}" }}")
    }

    fun getStats(): Result<DashboardStats> {
        return executeRequest<DashboardStats>("GET", "/calls/stats")
    }

    fun togglePin(callId: Int, pinned: Boolean): Result<Map<String, Any>> {
        return executeRequest<Map<String, Any>>("PUT", "/calls/$callId/pin?pinned=$pinned")
    }

    fun updateCallNotes(callId: Int, notes: String): Result<Map<String, Any>> {
        return executeRequest<Map<String, Any>>("PUT", "/calls/$callId/notes?notes=${notes.replace(" ", "%20")}")
    }

    // ─── Leads ───

    fun createLead(lead: Lead): Result<Lead> {
        val body = mapOf(
            "name" to lead.name,
            "phone" to lead.phone,
            "email" to (lead.email ?: ""),
            "status" to lead.status,
            "source" to (lead.source ?: ""),
            "priority" to lead.priority,
            "notes" to (lead.notes ?: ""),
            "assigned_to" to (lead.assigned_to ?: -1).toString()
        )
        return executeRequest<Lead>("POST", "/leads", body)
    }

    fun getLeads(status: String? = null, priority: String? = null,
                 search: String? = null, limit: Int = 200, offset: Int = 0): Result<List<Lead>> {
        val params = mutableMapOf<String, String>("limit" to limit.toString(), "offset" to offset.toString())
        status?.let { params["status"] = it }
        priority?.let { params["priority"] = it }
        search?.let { params["search"] = it }
        return executeRequest<List<Lead>>("GET", "/leads?${params.entries.joinToString("&") { "${it.key}=${it.value}" }}")
    }

    fun getLead(id: Int): Result<Lead> {
        return executeRequest<Lead>("GET", "/leads/$id")
    }

    fun updateLead(id: Int, lead: Lead): Result<Lead> {
        val body = mapOf(
            "name" to lead.name,
            "phone" to lead.phone,
            "email" to (lead.email ?: ""),
            "status" to lead.status,
            "source" to (lead.source ?: ""),
            "priority" to lead.priority,
            "notes" to (lead.notes ?: "")
        )
        return executeRequest<Lead>("PUT", "/leads/$id", body)
    }

    fun markLeadContacted(id: Int): Result<Map<String, Any>> {
        return executeRequest<Map<String, Any>>("POST", "/leads/$id/contact")
    }

    // ─── Reports ───

    fun getDailyReports(days: Int = 30): Result<List<DailyReportData>> {
        return executeRequest<List<DailyReportData>>("GET", "/reports/daily?days=$days")
    }

    fun generateReport(): Result<Map<String, Any>> {
        return executeRequest<Map<String, Any>>("POST", "/reports/generate-email", null)
    }

    // ─── Team ───

    fun getUsers(): Result<List<User>> {
        return executeRequest<List<User>>("GET", "/users")
    }

    fun updateUserRole(userId: Int, role: String): Result<Map<String, Any>> {
        return executeRequest<Map<String, Any>>("POST", "/users/$userId/role?role=$role")
    }

    // ─── Helpers ───

    private fun <T> executeRequest(
        method: String,
        endpoint: String,
        body: Map<String, Any>? = null
    ): Result<T> {
        return try {
            val requestBody = if (body != null) {
                val json = gson.toJson(body)
                RequestBody.create(json, "application/json".toMediaType())
            } else null

            val request = Request.Builder()
                .url("$baseUrl$endpoint")
                .method(method, requestBody)
                .build()

            val response = client.newCall(request).execute()
            if (response.isSuccessful) {
                val responseBody = response.body?.string()
                if (responseBody != null) {
                    val result = gson.fromJson(responseBody, object : com.google.gson.reflect.TypeToken<T>() {}.type)
                    Result.success(result)
                } else {
                    Result.success(null as T)
                }
            } else {
                val errorBody = response.body?.string() ?: "Unknown error"
                val errorJson = gson.fromJson(errorBody, Map::class.java)
                val detail = errorJson["detail"] as? String ?: errorBody
                Result.failure(Exception(detail))
            }
        } catch (e: Exception) {
            Result.failure(e)
        }
    }
}

class LoggingInterceptor : Interceptor {
    override fun intercept(chain: Chain): Response {
        val request = chain.request()
        println("API REQUEST: ${request.method} ${request.url}")
        val startTime = System.currentTimeMillis()
        val response = chain.proceed(request)
        val elapsed = System.currentTimeMillis() - startTime
        println("API RESPONSE: ${response.code} in ${elapsed}ms")
        return response
    }
}

package com.levoy.app.network

import android.content.Context
import android.content.SharedPreferences

class TokenStorage(context: Context) {
    private val prefs: SharedPreferences = context.getSharedPreferences("levoy_prefs", Context.MODE_PRIVATE)

    companion object {
        private const val KEY_TOKEN = "auth_token"
        private const val KEY_USER = "user_json"
        private const val KEY_USER_ID = "user_id"
        private const val KEY_USER_NAME = "user_name"

        @Volatile
        private var instance: TokenStorage? = null

        fun getInstance(context: Context): TokenStorage {
            return instance ?: synchronized(this) {
                instance ?: TokenStorage(context).also { instance = it }
            }
        }
    }

    fun getToken(): String? = prefs.getString(KEY_TOKEN, null)

    fun saveToken(token: String) = prefs.edit().putString(KEY_TOKEN, token).apply()

    fun clearToken() = prefs.edit().remove(KEY_TOKEN).apply()

    fun saveUser(userId: Int, name: String, email: String) {
        prefs.edit()
            .putInt(KEY_USER_ID, userId)
            .putString(KEY_USER_NAME, name)
            .putString(KEY_USER_EMAIL, email)
            .apply()
    }

    fun getUser(): UserInfo? {
        val id = prefs.getInt(KEY_USER_ID, -1)
        if (id == -1) return null
        return UserInfo(
            id = id,
            name = prefs.getString(KEY_USER_NAME, "Unknown") ?: "Unknown",
            email = prefs.getString(KEY_USER_EMAIL, "") ?: ""
        )
    }

    fun clearAll() = prefs.edit().clear().apply()
}

data class UserInfo(val id: Int, val name: String, val email: String)

package com.levoy.app.calllog

import android.content.ContentObserver
import android.content.Context
import android.database.Cursor
import android.net.Uri
import android.os.Handler
import android.os.Looper
import com.levoy.app.model.CallEvent
import com.levoy.app.model.CallEventUpload
import com.levoy.app.network.ApiClient
import com.levoy.app.network.TokenStorage
import java.text.SimpleDateFormat
import java.util.*

/**
 * Core class: reads Android call log and uploads to backend.
 *
 * Strategy: Uses ContentObserver on CallLog.Calls content URI to get real-time
 * notifications of new/changed calls, then fetches the call details from the
 * content provider.
 *
 * Fallback: If ContentObserver isn't sufficient, uses WorkManager periodic sync
 * to poll for new calls since last sync.
 */
class CallLogTracker private constructor(private val context: Context) {

    companion object {
        @Volatile
        private var instance: CallLogTracker? = null

        fun getInstance(context: Context): CallLogTracker {
            return instance ?: synchronized(this) {
                instance ?: CallLogTracker(context.applicationContext).also { instance = it }
            }
        }
    }

    private val api: ApiClient = ApiClient(context)
    private val tokenStorage = TokenStorage.getInstance(context)
    private val dateFormat = SimpleDateFormat("yyyy-MM-dd'T'HH:mm:ss", Locale.US)

    // Track last sync timestamp to avoid re-uploading
    private var lastSyncTimestamp: Long = 0L

    private val callLogObserver = CallLogContentObserver(Handler(Looper.getMainLooper()))

    fun startTracking() {
        context.contentResolver.registerContentObserver(
            Uri.parse("content://com.android.contacts/call_log"),
            false,
            callLogObserver
        )
        // Also observe the standard call log URI
        context.contentResolver.registerContentObserver(
            Uri.parse("content://call_log/calls"),
            false,
            callLogObserver
        )
        // Initial sync of existing calls
        syncExistingCalls()
    }

    fun stopTracking() {
        context.contentResolver.unregisterContentObserver(callLogObserver)
    }

    /**
     * Called when call log changes. Picks up new/modified calls and queues them.
     */
    private fun onCallLogChanged() {
        val pendingCalls = fetchNewCalls()
        if (pendingCalls.isEmpty()) return
        // Queue for background upload
        CallUploadQueue.getInstance(context).addCalls(pendingCalls)
        // Try immediate upload if online
        if (tokenStorage.getToken() != null) {
            uploadPendingCalls()
        }
    }

    /**
     * Fetch calls newer than lastSyncTimestamp.
     * Uses the Android CallLog.Calls content provider.
     */
    private fun fetchNewCalls(): List<CallEventUpload> {
        val calls = mutableListOf<CallEventUpload>()
        val cursor: Cursor? = try {
            context.contentResolver.query(
                Uri.parse("content://call_log/calls"),
                arrayOf(
                    CallsColumns._ID,
                    CallsColumns.NUMBER,
                    CallsColumns.DATE,
                    CallsColumns.DURATION,
                    CallsColumns.TYPE,
                    CallsColumns.NAME,
                    CallsColumns.STATUS
                ),
                "${CallsColumns.DATE} > ?",
                arrayOf(lastSyncTimestamp.toString()),
                "${CallsColumns.DATE} DESC"
            )
        } catch (e: SecurityException) {
            // Permission not granted
            Log.e("CallLogTracker", "READ_CALL_LOG permission not granted", e)
            null
        }

        cursor?.use {
            val idCol = it.getColumnIndex(CallsColumns._ID)
            val numberCol = it.getColumnIndex(CallsColumns.NUMBER)
            val dateCol = it.getColumnIndex(CallsColumns.DATE)
            val durationCol = it.getColumnIndex(CallsColumns.DURATION)
            val typeCol = it.getColumnIndex(CallsColumns.TYPE)
            val nameCol = it.getColumnIndex(CallsColumns.NAME)
            val statusCol = it.getColumnIndex(CallsColumns.STATUS)

            while (it.moveToNext()) {
                val id = it.getLong(idCol)
                val number = it.getString(numberCol) ?: ""
                val date = it.getLong(dateCol)
                val duration = it.getInt(durationCol)
                val type = it.getInt(typeCol) // 1=incoming, 2=outgoing, 3=missed, 5=rejected
                val name = it.getString(nameCol)
                val status = if (statusCol >= 0) it.getInt(statusCol) else 0

                // Map Android call type to our direction/status
                val (direction, callStatus) = mapCallType(type, status)

                val event = CallEventUpload(
                    phone_number = normalizePhoneNumber(number),
                    direction = direction,
                    status = callStatus,
                    duration_seconds = duration,
                    timestamp = dateFormat.format(Date(date)),
                    caller_id = name,
                    is_pinned = false
                )
                calls.add(event)

                // Update last sync timestamp
                if (date > lastSyncTimestamp) {
                    lastSyncTimestamp = date
                }
            }
        }
        return calls
    }

    /**
     * Maps Android call log types/statuses to our schema.
     *
     * Android Types:
     *   INCOMING = 1
     *   OUTGOING = 2
     *   MISSED = 3
     *   VOICEMAIL = 4
     *   REJECTED = 5
     *   BLOCKED = 6
     *   ANSWER = 7
     *
     * Status values from CallLog.Calls.STATUS (Android 5+):
     *   IDLE = 0
     *   IN_PROGRESS = 1
     *   OFFHOOK = 2
     *   RUNNING = 3
     *   UNKNOWN = 4
     *   CALL_STATUS_MISSED = 5
     *   CALL_STATUS_NO_ANSWER = 6
     *   CALL_STATUS_REJECTED = 7
     *   CALL_STATUS_COMPLETED = 8
     */
    private fun mapCallType(type: Int, status: Int): Pair<String, String> {
        val direction = when (type) {
            CallsColumns.TYPE_INCOMING -> "incoming"
            CallsColumns.TYPE_OUTGOING -> "outgoing"
            CallsColumns.TYPE_MISSED -> "incoming"  // missed is still incoming direction
            CallsColumns.TYPE_REJECTED -> "outgoing" // rejected is outgoing direction
            CallsColumns.TYPE_BLOCKED -> "outgoing"
            else -> "outgoing"
        }

        val callStatus = when {
            type == CallsColumns.TYPE_MISSED -> "missed"
            type == CallsColumns.TYPE_REJECTED -> "rejected"
            type == CallsColumns.TYPE_BLOCKED -> "rejected"
            status == CallsColumns.CALL_STATUS_MISSED -> "missed"
            status == CallsColumns.CALL_STATUS_NO_ANSWER -> "no_answer"
            status == CallsColumns.CALL_STATUS_REJECTED -> "rejected"
            else -> "connected"
        }

        return Pair(direction, callStatus)
    }

    private fun normalizePhoneNumber(number: String): String {
        // Remove extensions, special chars, keep digits and +
        return number.replace(Regex("[^\\d+]"), "").take(20)
    }

    /**
     * Syncs all existing calls from the device (first-time setup).
     */
    fun syncExistingCalls() {
        lastSyncTimestamp = 0L
        val calls = fetchNewCalls()
        if (calls.isNotEmpty()) {
            CallUploadQueue.getInstance(context).addCalls(calls)
            uploadPendingCalls()
        }
    }

    /**
     * Uploads queued calls to the backend.
     */
    fun uploadPendingCalls(): Int {
        val queue = CallUploadQueue.getInstance(context)
        val pending = queue.getPendingCalls()
        if (pending.isEmpty()) return 0
        if (tokenStorage.getToken() == null) {
            // Not logged in — can't upload
            return 0
        }

        val result = api.uploadCallBatch(pending)
        if (result.isSuccess) {
            queue.markAsSynced(pending.size)
            val stats = result.getOrNull()
            stats?.let {
                android.util.Log.d("CallLogTracker", "Uploaded ${it["received"]} calls")
            }
            return pending.size
        } else {
            android.util.Log.e("CallLogTracker", "Upload failed: ${result.exceptionOrNull()?.message}")
            return 0
        }
    }

    /**
     * Periodic full sync (called by WorkManager).
     */
    fun fullSync(): Int {
        // Re-sync everything since lastSyncTimestamp
        val calls = fetchNewCalls()
        if (calls.isNotEmpty()) {
            CallUploadQueue.getInstance(context).addCalls(calls)
        }
        return uploadPendingCalls()
    }
}

package com.levoy.app.calllog

import android.content.Context
import android.content.SharedPreferences
import com.levoy.app.model.CallEventUpload
import java.util.concurrent.ConcurrentLinkedQueue

/**
 * In-memory + persisted queue of call events waiting to be uploaded.
 * Persists to SharedPreferences for durability across app restarts.
 */
class CallUploadQueue private constructor(private val context: Context) {

    companion object {
        @Volatile
        private var instance: CallUploadQueue? = null

        fun getInstance(context: Context): CallUploadQueue {
            return instance ?: synchronized(this) {
                instance ?: CallUploadQueue(context.applicationContext).also { instance = it }
            }
        }
    }

    private val prefs = context.getSharedPreferences("call_queue", Context.MODE_PRIVATE)
    private val pendingCalls = ConcurrentLinkedQueue<CallEventUpload>()

    init {
        // Load persisted queue from disk
        val stored = prefs.getString("pending_calls", null)
        stored?.let {
            val list = com.google.gson.Gson().fromJson(it, Array<CallEventUpload>::class.java)
            list.forEach { pendingCalls.add(it) }
        }
    }

    fun addCalls(calls: List<CallEventUpload>) {
        calls.forEach { pendingCalls.add(it) }
        persistQueue()
    }

    fun getPendingCalls(): List<CallEventUpload> {
        return pendingCalls.toList()
    }

    fun markAsSynced(count: Int) {
        repeat(count) {
            pendingCalls.poll()
        }
        persistQueue()
    }

    fun clear() {
        pendingCalls.clear()
        prefs.edit().remove("pending_calls").apply()
    }

    private fun persistQueue() {
        val json = com.google.gson.Gson().toJson(pendingCalls.toList())
        prefs.edit().putString("pending_calls", json).apply()
    }

    fun size(): Int = pendingCalls.size
}

/**
 * Foreground service that keeps call log sync alive in background.
 * Shows a persistent notification: "Levoy is tracking calls".
 */
class CallSyncService : Service() {

    private var wakeLock: PowerManager.WakeLock? = null
    private val executor = Executors.newSingleThreadExecutor()
    private var isRunning = false

    override fun onCreate() {
        super.onCreate()
        createWakeLock()
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_UPLOAD_PENDING -> executor.execute { uploadPendingCalls() }
            ACTION_FULL_SYNC -> executor.execute { fullSync() }
            ACTION_STOP -> stopSelf()
        }
        startForeground(NOTIFICATION_ID, createNotification())
        if (!isRunning) {
            isRunning = true
            startPeriodicSync()
        }
        return START_STICKY
    }

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onDestroy() {
        super.onDestroy()
        isRunning = false
        wakeLock?.let { it.release(); wakeLock = null }
        executor.shutdown()
    }

    private fun createWakeLock() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            wakeLock = getSystemService(Context.POWER_SERVICE) as PowerManager
                .newWakeLock(PowerManager.PARTIAL_WAKE_LOCK, "Levoy::CallSyncLock")
                .apply { setReferenceCounted(false) }
        }
    }

    private fun createNotification(): Notification {
        val intent = Intent(this, MainActivity::class.java).apply {
            flags = Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TASK
        }
        val pendingIntent = PendingIntent.getActivity(
            this, 0, intent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        return NotificationCompat.Builder(this, LevoyApp.CHANNEL_SYNC)
            .setContentTitle("Levoy")
            .setContentText("Tracking your call activity")
            .setSmallIcon(android.R.drawable.stat_call_made)
            .setContentIntent(pendingIntent)
            .set Ongoing(true)
            .build()
    }

    private fun startPeriodicSync() {
        // Use WorkManager for periodic sync (more reliable than AlarmManager)
        // This is handled by CallSyncWorker — see below
    }

    private fun uploadPendingCalls() {
        if (TokenStorage.getInstance(this).getToken() == null) return
        val tracker = CallLogTracker.getInstance(this)
        val uploaded = tracker.uploadPendingCalls()
        android.util.Log.d("CallSyncService", "Uploaded $uploaded pending calls")
    }

    private fun fullSync() {
        val tracker = CallLogTracker.getInstance(this)
        val uploaded = tracker.fullSync()
        android.util.Log.d("CallSyncService", "Full sync: $uploaded calls uploaded")
    }

    companion object {
        const val NOTIFICATION_ID = 1001
        const val ACTION_UPLOAD_PENDING = "com.levoy.app.UPLOAD_PENDING"
        const val ACTION_FULL_SYNC = "com.levoy.app.FULL_SYNC"
        const val ACTION_STOP = "com.levoy.app.STOP"

        fun start(context: Context) {
            val intent = Intent(context, CallSyncService::class.java).apply {
                action = ACTION_UPLOAD_PENDING
            }
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
                context.startForegroundService(intent)
            } else {
                context.startService(intent)
            }
        }

        fun uploadPending(context: Context) {
            val intent = Intent(context, CallSyncService::class.java).apply {
                action = ACTION_UPLOAD_PENDING
            }
            context.startService(intent)
        }

        fun fullSync(context: Context) {
            val intent = Intent(context, CallSyncService::class.java).apply {
                action = ACTION_FULL_SYNC
            }
            context.startService(intent)
        }

        fun stop(context: Context) {
            val intent = Intent(context, CallSyncService::class.java).apply {
                action = ACTION_STOP
            }
            context.startService(intent)
        }
    }
}

package com.levoy.app.service

import android.content.Context
import androidx.work.*
import com.levoy.app.calllog.CallLogTracker
import com.levoy.app.network.TokenStorage
import java.util.concurrent.TimeUnit

/**
 * WorkManager worker: periodic background sync of call logs.
 * Runs every 15 minutes (or as configured).
 */
class CallSyncWorker(
    private val context: Context,
    workerParams: WorkerParameters
) : Worker(context, workerParams) {

    override fun doWork(): Result {
        val token = TokenStorage.getInstance(context).getToken()
        if (token == null) {
            // Not logged in — skip sync
            return Result.success()
        }

        try {
            val tracker = CallLogTracker.getInstance(context)
            val uploaded = tracker.fullSync()
            android.util.Log.d("CallSyncWorker", "Synced $uploaded call events")
            return Result.success()
        } catch (e: Exception) {
            android.util.Log.e("CallSyncWorker", "Sync failed", e)
            return Result.retry()
        }
    }

    companion object {
        private const val WORK_NAME = "call_sync_worker"

        fun schedule(context: Context) {
            val constraints = Constraints.Builder()
                .setRequiredNetworkType(NetworkType.CONNECTED)
                .build()

            val workRequest = PeriodicWorkRequestBuilder<CallSyncWorker>(
                15, TimeUnit.MINUTES,  // minimum interval for periodic work
                5, TimeUnit.MINUTES     // flex interval
            )
                .setConstraints(constraints)
                .setBackoffCriteria(BackoffPolicy.EXPONENTIAL, 1, TimeUnit.MINUTES)
                .setRequiresCharging(false)
                .build()

            WorkManager.getInstance(context).enqueueUniquePeriodicWork(
                WORK_NAME,
                ExistingPeriodicWorkPolicy.KEEP,
                workRequest
            )
        }

        fun cancel(context: Context) {
            WorkManager.getInstance(context).cancelUniqueWork(WORK_NAME)
        }
    }
}

        }
    }

        if (intent.action == TelephonyManager.ACTION_PHONE_STATE_CHANGED) {
            val state = intent.getStringExtra(TelephonyManager.EXTRA_STATE)
            when (state) {
                TelephonyManager.EXTRA_STATE_RINGING -> {
                    // Call coming in — we'll pick it up from call log after it ends
                }
                TelephonyManager.EXTRA_STATE_OFFHOOK -> {
                    // Call in progress
                }
                TelephonyManager.EXTRA_STATE_IDLE -> {
                    // Call ended — trigger call log sync
                    CallLogTracker.getInstance(context).onCallLogChanged()
                }
            }
        }
    }

package com.levoy.app.fragment

import android.os.Bundle
import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import android.widget.*
import androidx.fragment.app.Fragment
import com.levoy.app.R
import com.levoy.app.network.ApiClient
import com.levoy.app.network.TokenStorage

class LoginFragment : Fragment() {

    private var api: ApiClient? = null

    override fun onCreateView(inflater: LayoutInflater, container: ViewGroup?, savedInstanceState: Bundle?): View? {
        val view = inflater.inflate(R.layout.fragment_login, container, false)

        api = ApiClient(requireContext())

        view.findViewById<EditText>(R.id.etLoginEmail).setText("rajesh@company.com")
        view.findViewById<EditText>(R.id.etLoginPassword).setText("rajesh123")

        view.findViewById<Button>(R.id.btnLogin).setOnClickListener {
            val email = view.findViewById<EditText>(R.id.etLoginEmail).text.toString().trim()
            val password = view.findViewById<EditText>(R.id.etLoginPassword).text.toString().trim()

            if (email.isEmpty()) {
                Toast.makeText(context, "Enter email", Toast.LENGTH_SHORT).show()
                return@setOnClickListener
            }

            view.findViewById<ProgressBar>(R.id.progressLogin).visibility = View.VISIBLE
            view.findViewById<Button>(R.id.btnLogin).isEnabled = false

            // Try login first
            api?.login(email, password)?.onSuccess { resp ->
                TokenStorage.getInstance(requireContext()).saveToken(resp.access_token)
                TokenStorage.getInstance(requireContext()).saveUser(resp.user.id, resp.user.name, resp.user.email)
                val activity = requireActivity() as MainActivity
                activity.supportActionBar?.title = resp.user.name
                activity.loadDashboard()
            }?.onFailure { loginError ->
                // If login fails, try register
                if (loginError.message?.contains("Invalid") == true) {
                    api?.register(email, password, email.substringBefore("@"), null)?.onSuccess { resp ->
                        TokenStorage.getInstance(requireContext()).saveToken(resp.access_token)
                        TokenStorage.getInstance(requireContext()).saveUser(resp.user.id, resp.user.name, resp.user.email)
                        val activity = requireActivity() as MainActivity
                        activity.supportActionBar?.title = resp.user.name
                        activity.loadDashboard()
                    }?.onFailure { registerError ->
                        view.findViewById<ProgressBar>(R.id.progressLogin).visibility = View.GONE
                        view.findViewById<Button>(R.id.btnLogin).isEnabled = true
                        Toast.makeText(context, "Error: ${registerError.message}", Toast.LENGTH_LONG).show()
                    }
                } else {
                    view.findViewById<ProgressBar>(R.id.progressLogin).visibility = View.GONE
                    view.findViewById<Button>(R.id.btnLogin).isEnabled = true
                    Toast.makeText(context, "Error: ${loginError.message}", Toast.LENGTH_LONG).show()
                }
            }
        }

        view.findViewById<TextView>(R.id.tvRegisterLink).setOnClickListener {
            val email = view.findViewById<EditText>(R.id.etLoginEmail).text.toString().trim()
            if (email.isEmpty()) {
                Toast.makeText(context, "Enter email first", Toast.LENGTH_SHORT).show()
                return@setOnClickListener
            }
            val name = email.substringBefore("@").replace(Regex("[^a-zA-Z]"), " ").replace("\\s+".toRegex(), " ").trim()
            val password = view.findViewById<EditText>(R.id.etLoginPassword).text.toString().trim().takeIf { it.length >= 6 }
                ?: "password123"

            view.findViewById<ProgressBar>(R.id.progressLogin).visibility = View.VISIBLE
            view.findViewById<Button>(R.id.btnLogin).isEnabled = false

            api?.register(email, password, name)?.onSuccess { resp ->
                TokenStorage.getInstance(requireContext()).saveToken(resp.access_token)
                TokenStorage.getInstance(requireContext()).saveUser(resp.user.id, resp.user.name, resp.user.email)
                val activity = requireActivity() as MainActivity
                activity.supportActionBar?.title = resp.user.name
                activity.loadDashboard()
            }?.onFailure { error ->
                view.findViewById<ProgressBar>(R.id.progressLogin).visibility = View.GONE
                view.findViewById<Button>(R.id.btnLogin).isEnabled = true
                Toast.makeText(context, "Error: ${error.message}", Toast.LENGTH_LONG).show()
            }
        }

        return view
    }

    companion object {
        fun newInstance() = LoginFragment()
    }
}
